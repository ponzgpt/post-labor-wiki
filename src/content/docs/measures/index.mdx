import Sparkline from '../../../components/Sparkline.astro';
import { fred, ilo, byArea, householdIncome, rebase, real, pfd } from '../../../lib/data.mjs';

export const prod = rebase(fred('OPHNFB.csv'), 1973);
export const pay = rebase(fred('COMPRNFB.csv'), 1973);
export const gap = prod.map((p) => ({ x: p.x, v: p.v - pay.find((q) => q.x === p.x).v }));
export const h = householdIncome();
export const cpi = fred('CPIAUCSL.csv');
export const mfg = fred('MANEMP.csv').map((p) => ({ x: p.x, v: p.v / 1000 }));

If machines were taking over paid work, these lines would move first: labour's slice of income down, ownership and transfers up, fewer prime-age adults in work, fewer hours. Each tile shows the latest annual value, the change over its full history and a sparkline; follow the title for the full chart and the debate.

Snapshots are refreshed with `npm run data` and committed after review; the date on each tile is when its file was downloaded. The capital and transfer tiles use national accounts, which include imputed items, so they are a rough proxy for household cash income, not Shapiro's own measures.
